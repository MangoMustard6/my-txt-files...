magick hald:6 -colorspace yuv -fx "angle=180*pi/180; channel(u,.5+(u.g-.5)*cos(angle)-(u.b-.5)*sin(angle),.5+(u.g-.5)*sin(angle)+(u.b-.5)*cos(angle))" -colorspace srgb -modulate 100,100,1300 idk.ppm

hh=$(ffprobe -v error -select_streams v:0 -show_entries stream=height -of default=nw=1:nk=1 "$FILE_1")

total_dur=$(awk -v t="$t" 'BEGIN {print t * 16}')

# FIX: Remove -ss 0 from input creation entirely to avoid input/output duration conflicts on short files
ffmpeg -y -stream_loop -1 -i "$FILE_1" -t "$total_dur" -vf scale=640:360,setsar=1:1,fps=29.97 -c:v ffv1 -c:a pcm_s16le modfps.avi

w=$(ffprobe -v error -select_streams v:0 -show_entries stream=width -of default=nw=1:nk=1 modfps.avi)
h=$(ffprobe -v error -select_streams v:0 -show_entries stream=height -of default=nw=1:nk=1 modfps.avi)

ffmpeg -y -i modfps.avi -ss 0 -t "$t" -c:v ffv1 -c:a pcm_s16le 1.avi
ffmpeg -y -i modfps.avi -ss $t -t "$t" -stream_loop -1 -i ./displacemaps/tvsimulator.mov -filter_complex "movie=idk.ppm[h];[0][h]haldclut,crop=iw/2:ih:0:0,split[left][tmp];[tmp]hflip[right];[left][right]hstack,format=yuv420p,format=bgr32[00];[1]crop=iw:ih/1:0:0,scale=$w:$h,eq=contrast=0.375,format=bgr32,hue=b=-0.033[x];nullsrc=1x1,geq=r=128:g=128:b=128,scale=$w:$h,format=bgr32[y];[00][x][y]displace=edge=wrap[v]" -map "[v]" -map 0:a -pix_fmt yuv420p -t "$t" -c:v ffv1 -c:a pcm_s16le 2.avi

ffmpeg -y -i modfps.avi -ss $dur3 -t "$t" -vf "movie=hslhue_45_200.ppm,[in]haldclut,format=yuv444p,rotate=t*45*PI/180/$t,geq=p((W/2)+(X-(W/2))/lerp(1\,2\,T/$t)\,(H/2)+(Y-(H/2))/lerp(1\,2\,T/$t)),scale=$w:$h,crop=$w:$h,format=yuv420p" -af "rubberband=pitch=2^(1/12):window=short:transients=mixed:detector=soft:channels=together:pitchq=consistency" -c:v ffv1 -c:a pcm_s16le 4.avi

ffmpeg -y -i modfps.avi -ss $dur5 -t "$t" -vf "movie=hslhue_-54.ppm,[in]haldclut,format=yuv420p" -af "rubberband=pitch=2^(-1/12):window=short:transients=smooth:detector=soft:channels=together:pitchq=consistency" -c:v ffv1 -c:a pcm_s16le 6.avi
ffmpeg -y -i modfps.avi -ss $dur6 -t "$t" -vf "movie=hslhue_54.ppm,[in]haldclut,format=yuv420p" -af "rubberband=pitch=2^(1/12):window=short:transients=mixed:detector=soft:channels=together:pitchq=consistency" -c:v ffv1 -c:a pcm_s16le 7.avi
ffmpeg -y -i modfps.avi -ss $dur7 -t "$t" -vf "movie=hslhue_45_200.ppm,[in]haldclut,format=yuv444p,rotate=t*45*PI/180/$t,geq=p((W/2)+(X-(W/2))/lerp(1\,2\,T/$t)\,(H/2)+(Y-(H/2))/lerp(1\,2\,T/$t)),scale=$w:$h,crop=$w:$h,format=yuv420p" -af "rubberband=pitch=2^(1/12):window=short:transients=mixed:detector=soft:channels=together:pitchq=consistency" -c:v ffv1 -c:a pcm_s16le 8.avi

ffmpeg -y -i modfps.avi -ss $dur10 -t "$t" -vf "movie=hslhue_54.ppm,[in]haldclut,format=yuv420p" -af "rubberband=pitch=2^(1/12):window=short:transients=mixed:detector=soft:channels=together:pitchq=consistency" -c:v ffv1 -c:a pcm_s16le 11.avi
ffmpeg -y -i modfps.avi -ss $dur11 -t "$t" -vf "movie=hslhue_108_30.ppm,[in]haldclut,hflip,format=yuv420p" -af "rubberband=pitch=2^(3/12):window=short:transients=mixed:detector=soft:channels=together:pitchq=consistency" -c:v ffv1 -c:a pcm_s16le 12.avi
ffmpeg -y -i modfps.avi -ss $dur12 -t "$t" -vf "movie=hslhue_-54.ppm,[in]haldclut,format=yuv420p" -af "rubberband=pitch=2^(-1/12):window=short:transients=mixed:detector=soft:channels=together:pitchq=consistency" -c:v ffv1 -c:a pcm_s16le 13.avi
ffmpeg -y -i modfps.avi -ss $dur13 -t "$t" -vf hflip -c:v ffv1 -c:a pcm_s16le 14.avi
ffmpeg -y -i modfps.avi -ss $dur14 -t "$t" -vf "movie=hslhue_54.ppm,[in]haldclut,format=yuv420p" -af "rubberband=pitch=2^(1/12):window=short:transients=mixed:detector=soft:channels=together:pitchq=consistency" -c:v ffv1 -c:a pcm_s16le 15.avi
ffmpeg -y -i modfps.avi -ss $dur15 -t "$t" -vf "movie=hslhue_108_30.ppm,[in]haldclut,hflip,format=yuv420p" -af "rubberband=pitch=2^(3/12):window=short:transients=mixed:detector=soft:channels=together:pitchq=consistency" -c:v ffv1 -c:a pcm_s16le 16.avi
