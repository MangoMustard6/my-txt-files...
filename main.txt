# Cleanup previous temporary files
rm -f *.mp4 *.ppm list.txt
mkdir -p ./displacemaps ./output

# Download displacement map & generate Hald CLUTs
curl -s "https://file.garden/aTXso15ukD3mnuPI/tv_sim_displacement_map.mov" -o "./displacemaps/tvsimulator.mov" >/dev/null 2>&1 &
magick hald:4 -modulate 100,100,130 hslhue_54.ppm >/dev/null 2>&1 &
magick hald:4 -modulate 100,100,270 hslhue_-54.ppm >/dev/null 2>&1 &
magick hald:4 -modulate 100,100,0 hslhue_180.ppm >/dev/null 2>&1 &
magick hald:4 -modulate 100,100,112 hslhue_22.ppm >/dev/null 2>&1 &
magick hald:4 -modulate 100,130,160 hslhue_108_30.ppm >/dev/null 2>&1 &
wait

vidlen=$(ffprobe -i $FILE_1 -show_entries format=duration -v quiet -of csv="p=0")
s={or:{arg:0}|1.85}
e={or:{arg:1}|0.85}
use_tempo="{lower:{arg:2}}"
target_len={or:{arg:3}|5}

t=$(awk -v e="$e" 'BEGIN{ printf "%.4f", e}')
t2=$(awk -v e="$e" 'BEGIN{ printf "%.4f", (e/2)}')
t5=$(awk -v s="$s" -v target_len="$target_len" 'BEGIN{ printf "%.4f", s+target_len}')

# Rubberband Configuration
R="window=short:transients=mixed:detector=soft:channels=together:pitchq=consistency"

if [ "$use_tempo" = "true" ]; then
  # Standard tempo modifications for $t clips
  f2="tempo=1.059:"
  f3="tempo=0.890:"
  
  # Doubled tempo modifications for $t2 half-duration clips
  f1_fast="tempo=2:"
  f2_fast="tempo=2.118:"
  f3_fast="tempo=1.780:"
  f6_fast="tempo=2:"
  f8_fast="tempo=2.778:"
  f15_fast="tempo=1.880:"
  f17_fast="tempo=1.587:"
  f22_fast="tempo=2:"
  f24_fast="tempo=2.1189:"
else
  # No extra tempo shifts for $t clips
  f2=""
  f3=""
  
  # Default all $t2 half-duration clips to 2x speed so audio fits seamlessly without truncation
  f1_fast="tempo=2:"
  f2_fast="tempo=2:"
  f3_fast="tempo=2:"
  f6_fast="tempo=2:"
  f8_fast="tempo=2:"
  f15_fast="tempo=2:"
  f17_fast="tempo=2:"
  f22_fast="tempo=2:"
  f24_fast="tempo=2:"
fi

# Audio filters for full duration ($t) clips
a1="rubberband=pitch=1:$R"
a2="rubberband=pitch=2^(1/12):${f2}$R"
a3="rubberband=pitch=2^(-2/12):${f3}$R"

# Audio filters for half duration ($t2) clips (guaranteed 2x tempo speed-up)
a1_fast="rubberband=pitch=1:${f1_fast}$R"
a2_fast="rubberband=pitch=2^(1/12):${f2_fast}$R"
a3_fast="rubberband=pitch=2^(-2/12):${f3_fast}$R"
a6_fast="rubberband=pitch=2^(2/12):${f6_fast}window=short:transients=smooth:detector=soft:channels=together:pitchq=consistency"
a8_fast="rubberband=pitch=2^(3/12):${f8_fast}$R"
a15_fast="rubberband=pitch=2^(-1/12):${f15_fast}$R"
a17_fast="rubberband=pitch=2^(-4/12):${f17_fast}$R"
a22_fast="rubberband=pitch=2^(12/12):${f22_fast}$R"
a24_fast="rubberband=pitch=2^(13/12):${f24_fast}$R"

C="-y -loglevel error -c:v libx264 -preset ultrafast -crf 22 -r 30 -c:a pcm_s16le -ar 44100 -pix_fmt yuv420p"

# Base clip
ffmpeg -i $FILE_1 -vf scale=640:360,setsar=1:1 -ss $s -to $t5 $C 0_long.mp4

w=$(ffprobe -v error -select_streams v:0 -show_entries stream=width -of default=nw=1:nk=1 0_long.mp4)
h=$(ffprobe -v error -select_streams v:0 -show_entries stream=height -of default=nw=1:nk=1 0_long.mp4)

# Full duration ($t) clips
ffmpeg -i 0_long.mp4 -af "$a1,apad,atrim=0:$t,asetpts=PTS-STARTPTS" -t $t $C 1.mp4
ffmpeg -i 0_long.mp4 -vf "movie=hslhue_54.ppm,[in]haldclut",format=yuv420p -af "$a2,apad,atrim=0:$t,asetpts=PTS-STARTPTS" -t $t $C 2.mp4
ffmpeg -i 0_long.mp4 -stream_loop -1 -i ./displacemaps/tvsimulator.mov -filter_complex "movie=hslhue_180.ppm[h];[0][h]haldclut,hflip,crop=iw/2:ih:0:0,split[left][tmp];[tmp]hflip[right];[left][right]hstack,format=yuv420p,format=bgr32[00];[1]crop=iw:ih/1:0:0,scale=$w:$h,eq=contrast=0.375,format=bgr32,hue=b=-0.033[x];nullsrc=1x1,geq=r=128:g=128:b=128,scale=$w:$h,format=bgr32[y];[00][x][y]displace=edge=wrap[v]" -af "$a3,apad,atrim=0:$t,asetpts=PTS-STARTPTS" -map "[v]" -map 0:a -t $t $C 3.mp4

# Half duration ($t2) stutter clips (with mapped fast tempo controls)
ffmpeg -i 0_long.mp4 -af "$a1_fast,apad,atrim=0:$t2,asetpts=PTS-STARTPTS" -t $t2 $C 5.mp4
ffmpeg -i 0_long.mp4 -vf "movie=hslhue_22.ppm,[in]haldclut",hflip,format=yuv420p -af "$a6_fast,apad,atrim=0:$t2,asetpts=PTS-STARTPTS" -t $t2 $C 6.mp4
ffmpeg -i 0_long.mp4 -vf "movie=hslhue_54.ppm,[in]haldclut",format=yuv420p -af "$a2_fast,apad,atrim=0:$t2,asetpts=PTS-STARTPTS" -t $t2 $C 7.mp4
ffmpeg -i 0_long.mp4 -vf "movie=hslhue_108_30.ppm,[in]haldclut,format=yuv420p,format=yuv444p,scale=640:640,geq='p(X-((sin((T*5*0+(0*15))+(Y/H)*(PI*0)))*(-15*0)),Y-((sin((T*5*0+(0.17946*15))+(X/W)*(PI*6.09)))*(-15*0.72)))',scale=$w:$h,setsar=1:1,format=yuv420p" -af "$a8_fast,apad,atrim=0:$t2,asetpts=PTS-STARTPTS" -t $t2 $C 8.mp4
ffmpeg -i 0_long.mp4 -stream_loop -1 -i ./displacemaps/tvsimulator.mov -filter_complex "movie=hslhue_180.ppm[h];[0][h]haldclut,hflip,crop=iw/2:ih:0:0,split[left][tmp];[tmp]hflip[right];[left][right]hstack,format=yuv420p,format=bgr32[00];[1]crop=iw:ih/1:0:0,scale=$w:$h,eq=contrast=0.375,format=bgr32,hue=b=-0.033[x];nullsrc=1x1,geq=r=128:g=128:b=128,scale=$w:$h,format=bgr32[y];[00][x][y]displace=edge=wrap[v]" -af "$a3_fast,apad,atrim=0:$t2,asetpts=PTS-STARTPTS" -map "[v]" -map 0:a -t $t2 $C 9.mp4
ffmpeg -i 0_long.mp4 -vf swapuv -af "$a1_fast,apad,atrim=0:$t2,asetpts=PTS-STARTPTS" -t $t2 $C 10.mp4
ffmpeg -i 0_long.mp4 -vf "movie=hslhue_-54.ppm,[in]haldclut",format=yuv420p -af "$a15_fast,apad,atrim=0:$t2,asetpts=PTS-STARTPTS" -t $t2 $C 15.mp4
ffmpeg -i 0_long.mp4 -vf "movie=hslhue_180.ppm,[in]haldclut",format=yuv420p,negate,format=yuv420p -af "$a17_fast,apad,atrim=0:$t2,asetpts=PTS-STARTPTS" -t $t2 $C 17.mp4
ffmpeg -i 0_long.mp4 -vf negate,hflip -af "$a22_fast,apad,atrim=0:$t2,asetpts=PTS-STARTPTS" -t $t2 $C 22.mp4
ffmpeg -i 0_long.mp4 -vf "movie=hslhue_54.ppm,[in]haldclut",format=yuv420p,negate,hflip,format=yuv420p -af "$a24_fast,apad,atrim=0:$t2,asetpts=PTS-STARTPTS" -t $t2 $C 24.mp4
ffmpeg -i 0_long.mp4 -vf "movie=hslhue_108_30.ppm,[in]haldclut,format=yuv420p,format=yuv444p,scale=640:640,geq='p(X-((sin((T*5*0+(0*15))+(Y/H)*(PI*0)))*(-15*0)),Y-((sin((T*5*0+(0.17946*15))+(X/W)*(PI*6.09)))*(-15*0.72)))',scale=$w:$h,setsar=1:1,format=yuv420p" -af "$a8_fast,apad,atrim=0:$target_len,asetpts=PTS-STARTPTS" -t $target_len $C 28.mp4

cat << 'EOF' > list.txt

EOF

ffmpeg -y -loglevel error -f concat -safe 0 -i list.txt -c:v libx264 -preset ultrafast -crf 22 -c:a aac -ar 44100 -pix_fmt yuv420p -movflags +faststart ./output/pwhatextended.mp4

# Clean up temporary files
rm -f *.mp4 *.ppm list.txt
}
