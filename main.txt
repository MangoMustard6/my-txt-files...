# Cleanup previous temporary files

# Download displacement map & generate Hald CLUTs
curl -s "https://file.garden/aTXso15ukD3mnuPI/tv_sim_displacement_map.mov" -o "./displacemaps/tvsimulator.mov" >/dev/null 2>&1 &
magick hald:4 -modulate 100,100,130 hslhue_54.ppm >/dev/null 2>&1 &

t2=$(awk -v e="$e" 'BEGIN{ printf "%.4f", (e/2)}')
t3=$(awk -v s="$s" -v e="$e" 'BEGIN{ printf "%.4f", s+e}')
t5=$(awk -v s="$s" -v target_len="$target_len" 'BEGIN{ printf "%.4f", s+target_len}')

if [ "$use_tempo" = "true" ]; then
  a2="asetrate=46722,aresample=44100,atempo=0.94387"
  a3="asetrate=39289,aresample=44100,atempo=1.12246"
  a6="asetrate=49501,aresample=44100,atempo=0.89090"
  a8="asetrate=52445,aresample=44100,atempo=0.84090"
  a15="asetrate=41625,aresample=44100,atempo=1.05946"
  a17="asetrate=35002,aresample=44100,atempo=1.25992"
  a22="asetrate=88200,aresample=44100,atempo=0.5"
  a24="asetrate=93445,aresample=44100,atempo=0.47194"
else
  a2="asetrate=46722,aresample=44100"
  a3="asetrate=39289,aresample=44100"
  a6="asetrate=49501,aresample=44100"
  a8="asetrate=52445,aresample=44100"
  a15="asetrate=41625,aresample=44100"
  a17="asetrate=35002,aresample=44100"
  a22="asetrate=88200,aresample=44100"
  a24="asetrate=93445,aresample=44100"
fi

C="-y -loglevel error -c:v libx264 -preset ultrafast -crf 22 -r 30 -c:a aac -ar 44100 -pix_fmt yuv420p"

# Create Base Cuts with Frame-Accurate Output Seeking
ffmpeg -i $FILE_1 -vf scale=640:360,setsar=1:1 -ss $s -to $t3 $C 0.mp4
ffmpeg -i $FILE_1 -vf scale=640:360,setsar=1:1 -ss $s -to $t5 $C 0_long.mp4

w=$(ffprobe -v error -select_streams v:0 -show_entries stream=width -of default=nw=1:nk=1 0.mp4)
h=$(ffprobe -v error -select_streams v:0 -show_entries stream=height -of default=nw=1:nk=1 0.mp4)

ffmpeg -i 0.mp4 -t $t $C 1.mp4

ffmpeg -i 0.mp4 -vf "movie=hslhue_54.ppm,[in]haldclut",format=yuv420p -af "$a2" -t $t $C 2.mp4
ffmpeg -i 0.mp4 -stream_loop -1 -i ./displacemaps/tvsimulator.mov -filter_complex "movie=hslhue_180.ppm[h];[0][h]haldclut,hflip,crop=iw/2:ih:0:0,split[left][tmp];[tmp]hflip[right];[left][right]hstack,format=yuv420p,format=bgr32[00];[1]crop=iw:ih/1:0:0,scale=$w:$h,eq=contrast=0.375,format=bgr32,hue=b=-0.033[x];nullsrc=1x1,geq=r=128:g=128:b=128,scale=$w:$h,format=bgr32[y];[00][x][y]displace=edge=wrap[v]" -af "$a3" -map "[v]" -map 0:a -t $t $C 3.mp4

ffmpeg -i 0.mp4 -t $t2 $C 5.mp4
ffmpeg -i 0.mp4 -vf "movie=hslhue_22.ppm,[in]haldclut",hflip,format=yuv420p -af "$a6" -t $t2 $C 6.mp4
ffmpeg -i 0.mp4 -vf "movie=hslhue_54.ppm,[in]haldclut",format=yuv420p -af "$a2" -t $t2 $C 7.mp4
ffmpeg -i 0.mp4 -vf "movie=hslhue_108_30.ppm,[in]haldclut,format=yuv420p,format=yuv444p,scale=640:640,geq='p(X-((sin((T*5*0+(0*15))+(Y/H)*(PI*0)))*(-15*0)),Y-((sin((T*5*0+(0.17946*15))+(X/W)*(PI*6.09)))*(-15*0.72)))',scale=$w:$h,setsar=1:1,format=yuv420p" -af "$a8" -t $t2 $C 8.mp4
ffmpeg -i 0.mp4 -stream_loop -1 -i ./displacemaps/tvsimulator.mov -filter_complex "movie=hslhue_180.ppm[h];[0][h]haldclut,hflip,crop=iw/2:ih:0:0,split[left][tmp];[tmp]hflip[right];[left][right]hstack,format=yuv420p,format=bgr32[00];[1]crop=iw:ih/1:0:0,scale=$w:$h,eq=contrast=0.375,format=bgr32,hue=b=-0.033[x];nullsrc=1x1,geq=r=128:g=128:b=128,scale=$w:$h,format=bgr32[y];[00][x][y]displace=edge=wrap[v]" -af "$a3" -map "[v]" -map 0:a -t $t2 $C 9.mp4
ffmpeg -i 0.mp4 -vf swapuv -t $t2 $C 10.mp4
ffmpeg -i 0.mp4 -vf "movie=hslhue_-54.ppm,[in]haldclut",format=yuv420p -af "$a15" -t $t2 $C 15.mp4
ffmpeg -i 0.mp4 -vf "movie=hslhue_180.ppm,[in]haldclut",format=yuv420p,negate,format=yuv420p -af "$a17" -t $t2 $C 17.mp4

ffmpeg -i 0.mp4 -vf "movie=hslhue_54.ppm,[in]haldclut",format=yuv420p,negate,hflip,format=yuv420p -af "$a24" -t $t2 $C 24.mp4
ffmpeg -i 0_long.mp4 -vf "movie=hslhue_108_30.ppm,[in]haldclut,format=yuv420p,format=yuv444p,scale=640:640,geq='p(X-((sin((T*5*0+(0*15))+(Y/H)*(PI*0)))*(-15*0)),Y-((sin((T*5*0+(0.17946*15))+(X/W)*(PI*6.09)))*(-15*0.72)))',scale=$w:$h,setsar=1:1,format=yuv420p" -af "$a8" -t $target_len $C 28.mp4

cat << 'EOF' > list.txt

ffmpeg -y -loglevel error -f concat -safe 0 -i list.txt -c:v libx264 -preset ultrafast -crf 22 -c:a aac -ar 44100 -pix_fmt yuv420p -movflags +faststart ./output/pwhatextended.mp4
