{js:
const { execSync } = require('child_process');
const fs = require('fs');

const FILE_1 = process.env.FILE_1 || '{iv}';

// Power & duration arguments
const power = parseInt('{arg:0}', 10) || 1;
const durationArg = '{arg:1}' && '{arg:1}' !== '{arg:1}' ? '{arg:1}' : '5';

// Capture args:2 directly
let extraArgs = '{args:2}';
if (extraArgs === '{args:2}' || !extraArgs.trim()) {
  extraArgs = '';
}

const run = (cmd) => execSync(cmd, { stdio: 'pipe' }).toString().trim();

try {
  // 1. Initial trim
  execSync(`ffmpeg -y -i "${FILE_1}" -vf "trim=end=${durationArg}" -af "atrim=end=${durationArg}" -c:v libx264 -c:a aac 1.mp4`);

  // 2. Loop using {args:2}
  for (let i = 2; i <= power + 1; i++) {
    const prev = i - 1;
    execSync(`ffmpeg -y -i ${prev}.mp4 ${extraArgs} -t ${durationArg} -preset ultrafast -crf 23 -pix_fmt yuv420p ${i}.mp4`);
  }

  // 3. Create concat list
  let listContent = '';
  for (let i = 2; i <= power + 1; i++) {
    listContent += `file '${i}.mp4'\n`;
  }
  fs.writeFileSync('list.txt', listContent);

  // 4. Output final file
  execSync(`ffmpeg -y -f concat -safe 0 -i list.txt -c:v libx264 -preset ultrafast -crf 23 -pix_fmt yuv420p -c:a aac ./output/ihtx.mp4`);
} catch (err) {
  console.error(err.stderr ? err.stderr.toString() : err.message);
}
}
